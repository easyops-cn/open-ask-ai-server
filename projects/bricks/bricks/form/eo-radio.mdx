表单单选构件。

```yaml preview
- brick: eo-radio
  properties:
    options:
      - Beijing
      - Shanghai
      - Guangzhou
      - Shenzhen
# -- YAML HEADING DELIMITER start (1nbbm8) --
form/eo-radio
# -- YAML HEADING DELIMITER end (1nbbm8) --
# -- YAML DELIMITER (1nbbm8) --
# <eo-radio id="brick-1"></eo-radio>
# 
# <script>
#   const brick_1 = document.getElementById("brick-1");
#   brick_1.options = ["Beijing", "Shanghai", "Guangzhou", "Shenzhen"];
# </script>
# 
```
