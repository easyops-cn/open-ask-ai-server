### With Form

  events:
    validate.success:
      - action: console.log
    values.change:
      - action: console.log
  children:
    - brick: eo-auto-complete
      properties:
        name: city
        label: 城市
        required: true
        options:
          - Beijing
          - Shanghai
          - Guangzhou
