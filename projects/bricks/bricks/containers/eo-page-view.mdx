```yaml preview
brick: eo-page-view

    height: 300px

      textContent: Sub-Sidebar

    children:
      - brick: eo-frame-breadcrumb
        slot: breadcrumb
        properties:
          breadcrumb:
            - text: Home
              to: /Home
            - text: Detail
              to: /Detail
            - text: List
              to: /List
```
