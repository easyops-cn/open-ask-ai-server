## Examples

### Basic

```yaml preview
- brick: eo-search-bar
  slots:
    start:
      bricks:
        - brick: eo-search
          properties:
            placeholder: 请输入
            clearable: true
            trim: true
      type: bricks
# -- YAML HEADING DELIMITER start (1nbbm8) --
containers/eo-search-bar/basic
# -- YAML HEADING DELIMITER end (1nbbm8) --
# -- YAML DELIMITER (1nbbm8) --
# <eo-search-bar>
#   <eo-search slot="start" placeholder="请输入" clearable trim></eo-search>
# </eo-search-bar>
# 
```

### End Slot

```yaml preview

  slots:
    end:
      bricks:
        - brick: eo-button

            type: primary
            icon:
              lib: antd
              icon: bell
        - brick: eo-button
          properties:
            icon:
              lib: antd
              icon: bell
      type: bricks

      bricks:
        - brick: eo-search

            placeholder: 请输入
```
